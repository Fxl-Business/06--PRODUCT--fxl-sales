## Objective

Review the Builder's diff, run the named commands, and return structured findings.

## Inputs

The diff, the acceptance criteria, the test and command output, and the previous round's findings (empty in round 1).

## Output

Findings JSON validated against [`../agents-findings.schema.json`](../agents-findings.schema.json): `{verdict, round, findings[], checks[]}`. `verdict` is `APPROVED` or `CHANGES_REQUESTED`. Each finding `kind` is `objective` (proven by a command) or `judgment` (a design opinion). A response that does not validate counts as a failed round, repeated once, then plano B.
On codex you answer the schema's strict variant, where an optional key such as `line` or `suggestion` is present and may be null.

## Writing area (D8)

You are given `FXL_SCRATCH`, a folder inside this run's scratchpad.
Your sandbox is read-only, so you write no file at all, not even inside `$FXL_SCRATCH`, which only holds inputs this run prepared, such as the diff under review.

You never edit code.
Being read-only, you cannot write even in `$FXL_SCRATCH`: read every command's output directly, and use the oracle and lint output the orchestrator already put in your prompt instead of re-running them.

## No approval, ever (D7)

You run non-interactively. You may never open a host approval prompt. An action that would require approval must fail immediately with a named error, never wait. /**
 * The read side of the CRM configuration: `GET /pipelines` and `GET /field-definitions`.
 * Both seed the org's defaults on first read (the default pipeline and the Cliente system field),
 * so a brand-new org answers a usable configuration. There is no write side in this run: pipelines,
 * stages and definitions are not editable through the API, which is what keeps the Cliente
 * definition impossible to remove or archive.
 */
type Db = ReturnType<typeof getDb>;

export type PipelineView = {
  id: string;
  name: string;
  isDefault: boolean;
  position: number;
  status: string;
  stages: Array<{
    id: string;
    name: string;
    position: number;
    type: string;
    behavior: string;
    exitKind: string | null;
    requiresReason: boolean;
    status: string;
  }>;
  transitions: Array<{ fromStageId: string; toStageId: string }>;
};

export function listPipelines(db: Db, orgId: string): Promise<PipelineView[]> {
  return withTenant(db, orgId, async (tx) => {
    await ensureCrmDefaults(tx, orgId);
    const pipelines = await tx
      .select()
      .from(salesOpsPipelines)
      .where(and(eq(salesOpsPipelines.orgId, orgId), eq(salesOpsPipelines.status, 'active')))
      .orderBy(asc(salesOpsPipelines.position), asc(salesOpsPipelines.name));
    const stages = await tx
      .select()
      .from(salesOpsPipelineStages)
      .where(eq(salesOpsPipelineStages.orgId, orgId))
      .orderBy(asc(salesOpsPipelineStages.position), asc(salesOpsPipelineStages.name));
    const transitions = await tx
      .select()
      .from(salesOpsPipelineTransitions)
      .where(eq(salesOpsPipelineTransitions.orgId, orgId));
    return pipelines.map((pipeline) => ({
      id: pipeline.id,
      name: pipeline.name,
      isDefault: pipeline.isDefault,
      position: pipeline.position,
      status: pipeline.status,
      stages: stages
        .filter((stage) => stage.pipelineId === pipeline.id)
        .map((stage) => ({
          id: stage.id,
          name: stage.name,
          position: stage.position,
          type: stage.type,
          behavior: stage.behavior,
          exitKind: stage.exitKind,
          requiresReason: stage.requiresReason,
          status: stage.status,
        })),
      transitions: transitions
        .filter((transition) => transition.pipelineId === pipeline.id)
        .map((transition) => ({ fromStageId: transition.fromStageId, toStageId: transition.toStageId })),
    }));
  });
}

/** The custom range the operator typed is reported when it is malformed, and never applied. */
export function periodError(selection: PeriodSelection): string | null {
  if (selection.preset !== 'custom') return null;
  const { from, to } = selection;
  if ((from !== '' && !isIsoDay(from)) || (to !== '' && !isIsoDay(to))) return PERIOD_COPY.badDay;
  if (from !== '' && to !== '' && from > to) return PERIOD_COPY.invalid;
  return null;
}

}

      );
    }
    const limit = query.limit ?? LEADS_DEFAULT_LIMIT;
    const rows = await tx
      .select({
        id: salesOpsCrmLeads.id,
        contactName: salesOpsCrmLeads.contactName,
        clientName: salesOpsCrmLeads.clientNameSnapshot,
        // The etapa's CURRENT name, joined live, so a renamed etapa reads renamed.
        stageName: salesOpsPipelineStages.name,
        sellerName: salesOpsCrmLeads.sellerNameSnapshot,
        estimatedValueBrl: salesOpsCrmLeads.estimatedValueBrl,
        deletedAt: salesOpsCrmLeads.deletedAt,
        deletedByName: salesOpsCrmLeads.deletedByName,
        // Rendered by Postgres with all six fractional digits: see the cursor
        // comment in lead-schemas.ts for why a JS Date cannot build it.
        cursorAt: sql<string>`to_char(${salesOpsCrmLeads.deletedAt} AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"')`,
      })
      .from(salesOpsCrmLeads)
      .innerJoin(
        salesOpsPipelineStages,
        and(
          eq(salesOpsPipelineStages.orgId, salesOpsCrmLeads.orgId),
          eq(salesOpsPipelineStages.id, salesOpsCrmLeads.stageId),
        ),
      )
      .where(and(...conditions))
      .orderBy(desc(salesOpsCrmLeads.deletedAt), desc(salesOpsCrmLeads.id))
      .limit(limit + 1);

    const hasMore = rows.length > limit;
    const page = hasMore ? rows.slice(0, limit) : rows;
    const last = page.at(-1);
    return {
      // Projected field by field: deleted_by_user_id is never selected, so it can
      // never reach a client.
      items: page.map((row) => ({
        id: row.id,
        contactName: row.contactName,
        clientName: row.clientName,
        stageName: row.stageName,
        sellerName: row.sellerName,
        estimatedValueBrl: row.estimatedValueBrl,
        deletedAt: row.deletedAt!.toISOString(),
        deletedByName: row.deletedByName,
      })),
      nextCursor: hasMore && last ? `${last.cursorAt}_${last.id}` : null,
    };
  });
}
