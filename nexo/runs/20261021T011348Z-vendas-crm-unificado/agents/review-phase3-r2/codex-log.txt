          clientName: current.clientNameSnapshot,
          stageName: fromStage,
          sellerName: current.sellerNameSnapshot,
        },
      },
    });
    return { ok: true } as const;
  });
}

/** Admins restore any org lead; others restore only their own deletion. */
export function restoreLead(
  db: Db,
  orgId: string,
  id: string,
  isAdmin: boolean,
  actor: LeadActor,
): Promise<RestoreLeadResult> {
  return withTenant(db, orgId, async (tx) => {
    const restoreConditions = and(
      eq(salesOpsCrmLeads.orgId, orgId),
      eq(salesOpsCrmLeads.id, id),
      deletedLeadCondition(),
      ...(isAdmin ? [] : [eq(salesOpsCrmLeads.deletedByUserId, actor.userId)]),
    );
    // A restore writes stage_id and position: same lock order as every board writer.
    await lockLeadBoard(tx, orgId);

    const [current] = await tx
      .select()
      .from(salesOpsCrmLeads)
      .where(restoreConditions)
      .limit(1)
      .for('update');
    // A live, unknown or other-org id: there is nothing to restore.
    if (!current) return { ok: false, reason: 'not_found' } as const;

    const [own] = await tx
      .select({
        id: salesOpsPipelineStages.id,
        name: salesOpsPipelineStages.name,
        status: salesOpsPipelineStages.status,
      })
      .from(salesOpsPipelineStages)
      .where(and(eq(salesOpsPipelineStages.orgId, orgId), eq(salesOpsPipelineStages.id, current.stageId)))
      .limit(1);
    // Its own etapa when it is still on the board; an archived etapa is not a
    // move target, so the lead goes where a new lead would. Never seeds one: the
    // leads edition may have none, and that is the existing 400.
    const target =
      own && own.status === 'active'
        ? { id: own.id, name: own.name }
        : await firstOpenLeadStage(tx, orgId, current.pipelineId);
    if (!target) throw new LeadInputError('no_open_stage');

    const stageChanged = target.id !== current.stageId;
    const position = await nextLeadPosition(tx, orgId, target.id);
    await tx
      .update(salesOpsCrmLeads)
      .set({
        deletedAt: null,
        deletedByUserId: null,
        deletedByName: null,
        stageId: target.id,
        position,
        // KEY OMISSION, the moveLead rule: stage_changed_at moves only when the
        // etapa changes. The fallback target is a normal etapa, which carries no
        // lost reason, so the reason is cleared with it.
        ...(stageChanged ? { stageChangedAt: sql`now()`, lostReason: null } : {}),
        updatedAt: new Date(),
      })
      .where(restoreConditions);
    // Appended at the end; the renumber keeps the column dense even if it was not.
    await renumberStage(tx, orgId, target.id, null);
    // A restore that lands in another etapa is a stage change like any other: one history event.
    if (stageChanged) {
      await recordStageChange(tx, {
        orgId,
        leadId: id,
        fromStageId: current.stageId,
        toStageId: target.id,
        reason: null,
        actorUserId: actor.userId,
        actorName: actor.name,
      });
    }

    const lead = await readLeadView(tx, orgId, id);
    // LAST: the ledger tail lock is global and held until COMMIT. A restore is a
    // NEW entry, never an undo of the delete entry.
    await writeAuditEntry(tx, {
      actorUserId: actor.userId,
      actorOrgId: orgId,
      action: 'lead.restored',
      entityType: 'lead',
      entityId: id,
      beforeJsonb: { deleted: true, stageName: own?.name ?? '' },
      afterJsonb: {
        deleted: false,
        label: current.contactName,
        actorLabel: actor.name,
        metadata: {
          contactName: current.contactName,
          clientName: current.clientNameSnapshot,
          stageName: target.name,
          sellerName: current.sellerNameSnapshot,
        },
      },
    });
    return { ok: true, lead } as const;
  });
}

/**

export async function getOrgModules(c: Context): Promise<OrgModules> {

}

function isModuleSource(value: string): value is CrmModuleSource {
  return (CRM_MODULE_SOURCES as readonly string[]).includes(value);

function stateFromRows(rows: readonly ModuleRow[]): OrgModulesState {
  const modules = {} as OrgModules;
  const sources = {} as OrgModuleSources;
  for (const key of CRM_MODULE_KEYS) {
    modules[key] = false;
    sources[key] = 'default';
  }
  for (const row of rows) {
    if (!isCrmModuleKey(row.moduleKey)) continue;
    modules[row.moduleKey] = row.enabled;
    sources[row.moduleKey] = isModuleSource(row.source) ? row.source : 'default';

  return { modules, sources };
}

async function readRows(tx: Db, orgId: string): Promise<ModuleRow[]> {
  return tx.select().from(salesOpsOrgModules).where(eq(salesOpsOrgModules.orgId, orgId));
}

/**
 * Returns the org's rows, writing the one-time compat backfill first when it has none.
 * MUST run inside a tenant transaction.
 */
async function ensureRows(
  tx: Db,
  orgId: string,
  tokenModules: readonly unknown[] | null | undefined,
): Promise<ModuleRow[]> {
  const existing = await readRows(tx, orgId);
  if (existing.length > 0) return existing;

  // Serialize first backfills of the same org: the loser blocks here, then re-reads.
  await tx.execute(sql`SELECT pg_advisory_xact_lock(hashtextextended(${`org-modules:${orgId}`}, 0))`);
  const again = await readRows(tx, orgId);
  if (again.length > 0) return again;

  const seed = modulesForEdition(resolveSalesEdition(tokenModules));
  await tx
    .insert(salesOpsOrgModules)
    .values(
      CRM_MODULE_KEYS.map((moduleKey) => ({
        orgId,
        moduleKey,
        enabled: seed[moduleKey],
        source: 'token_backfill' as const,
      })),
    )
    .onConflictDoNothing({ target: [salesOpsOrgModules.orgId, salesOpsOrgModules.moduleKey] });
  return readRows(tx, orgId);
}

export async function loadOrgModules(
  orgId: string,
  tokenModules: readonly unknown[] | null | undefined,
  db: Db = getDb(),
): Promise<OrgModulesState> {
  return withTenant(db, orgId, async (tx) => stateFromRows(await ensureRows(tx, orgId, tokenModules)));
}

/** Admin toggle: ensure the backfill, then upsert the patched keys as source 'admin'. */
export async function updateOrgModules(
  orgId: string,
  tokenModules: readonly unknown[] | null | undefined,
  patch: OrgModulesPatch,
  db: Db = getDb(),
): Promise<OrgModulesState> {
  return withTenant(db, orgId, async (tx) => {
    await ensureRows(tx, orgId, tokenModules);
    for (const key of CRM_MODULE_KEYS) {
      const enabled = patch[key];
      if (enabled === undefined) continue;
      await tx
        .insert(salesOpsOrgModules)
        .values({ orgId, moduleKey: key, enabled, source: 'admin', updatedAt: new Date() })
        .onConflictDoUpdate({
          target: [salesOpsOrgModules.orgId, salesOpsOrgModules.moduleKey],
          set: { enabled, source: 'admin', updatedAt: new Date() },
        });
    }
    return stateFromRows(await readRows(tx, orgId));
  });
