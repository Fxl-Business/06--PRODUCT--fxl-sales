   ```sql
   (SELECT c."nm" FROM (
      SELECT g.k, CASE WHEN g.k = 0 THEN e."label" WHEN g.k = 1 THEN e."label" || ' (saída)' ELSE e."label" || ' (saída ' || g.k || ')' END AS "nm"

   ```
